import React, { useState, useEffect } from 'react';
import { Card, CardHeader, CardTitle, CardContent } from '@/components/ui/card';
import { Input } from '@/components/ui/input';
import { Button } from '@/components/ui/button';
import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from '@/components/ui/table';
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select';

const OptionsChainSimulator = () => {
  const [ticker, setTicker] = useState('SPY');
  const [expirationDates, setExpirationDates] = useState([]);
  const [selectedExpiration, setSelectedExpiration] = useState('');
  const [optionsChain, setOptionsChain] = useState([]);
  const [underlyingPrice, setUnderlyingPrice] = useState(0);
  const [positions, setPositions] = useState([]);
  const [balance, setBalance] = useState(10000);
  const [loading, setLoading] = useState(false);

  // Replace with your Polygon API key
  const POLYGON_API_KEY = 'YOUR_POLYGON_API_KEY';

  // Fetch available expiration dates
  useEffect(() => {
    const fetchExpirationDates = async () => {
      try {
        setLoading(true);
        const response = await fetch(
          `https://api.polygon.io/v3/reference/options/contracts?underlying_ticker=${ticker}&apiKey=${POLYGON_API_KEY}`
        );
        const data = await response.json();
        
        // Extract unique expiration dates
        const dates = [...new Set(data.results.map(contract => contract.expiration_date))];
        dates.sort();
        setExpirationDates(dates);
        
        if (dates.length > 0) {
          setSelectedExpiration(dates[0]);
        }
      } catch (error) {
        console.error('Error fetching expiration dates:', error);
      } finally {
        setLoading(false);
      }
    };

    fetchExpirationDates();
  }, [ticker]);

  // Fetch options chain data
  useEffect(() => {
    const fetchOptionsChain = async () => {
      if (!selectedExpiration) return;

      try {
        setLoading(true);
        
        // Fetch underlying price
        const stockResponse = await fetch(
          `https://api.polygon.io/v2/last/trade/${ticker}?apiKey=${POLYGON_API_KEY}`
        );
        const stockData = await stockResponse.json();
        setUnderlyingPrice(stockData.results.p);

        // Fetch options chain
        const optionsResponse = await fetch(
          `https://api.polygon.io/v3/snapshot/options/${ticker}?expiration_date=${selectedExpiration}&apiKey=${POLYGON_API_KEY}`
        );
        const optionsData = await optionsResponse.json();

        // Process and format the options data
        const processedChain = optionsData.results.reduce((acc, contract) => {
          const strike = contract.strike_price;
          const existingStrike = acc.find(item => item.strike === strike);

          if (existingStrike) {
            if (contract.call_put === 'call') {
              existingStrike.call = {
                bid: contract.bid,
                ask: contract.ask,
                last: contract.last_trade,
                volume: contract.volume,
                openInterest: contract.open_interest
              };
            } else {
              existingStrike.put = {
                bid: contract.bid,
                ask: contract.ask,
                last: contract.last_trade,
                volume: contract.volume,
                openInterest: contract.open_interest
              };
            }
          } else {
            acc.push({
              strike,
              call: contract.call_put === 'call' ? {
                bid: contract.bid,
                ask: contract.ask,
                last: contract.last_trade,
                volume: contract.volume,
                openInterest: contract.open_interest
              } : {},
              put: contract.call_put === 'put' ? {
                bid: contract.bid,
                ask: contract.ask,
                last: contract.last_trade,
                volume: contract.volume,
                openInterest: contract.open_interest
              } : {}
            });
          }
          return acc;
        }, []);

        processedChain.sort((a, b) => a.strike - b.strike);
        setOptionsChain(processedChain);
      } catch (error) {
        console.error('Error fetching options chain:', error);
      } finally {
        setLoading(false);
      }
    };

    fetchOptionsChain();
  }, [ticker, selectedExpiration]);

  const executeOrder = (strike, type, isBuy) => {
    const option = optionsChain.find(opt => opt.strike === strike);
    const optionData = type === 'call' ? option.call : option.put;
    const price = isBuy ? optionData.ask : optionData.bid;
    
    const cost = price * 100; // Each contract represents 100 shares
    
    if (isBuy && cost > balance) {
      alert('Insufficient funds!');
      return;
    }

    setBalance(prev => isBuy ? prev - cost : prev + cost);
    setPositions(prev => [...prev, {
      type,
      strike,
      price,
      isBuy,
      timestamp: new Date().toLocaleString()
    }]);
  };

  return (
    <div className="space-y-4 p-4">
      <Card>
        <CardHeader>
          <CardTitle>Live Options Chain Trading Simulator</CardTitle>
        </CardHeader>
        <CardContent>
          <div className="flex items-center gap-4 mb-4">
            <div>
              <label className="block text-sm font-medium mb-1">
                Ticker
              </label>
              <Input 
                value={ticker}
                onChange={(e) => setTicker(e.target.value.toUpperCase())}
                className="w-32"
              />
            </div>
            <div>
              <label className="block text-sm font-medium mb-1">
                Expiration Date
              </label>
              <Select value={selectedExpiration} onValueChange={setSelectedExpiration}>
                <SelectTrigger className="w-48">
                  <SelectValue />
                </SelectTrigger>
                <SelectContent>
                  {expirationDates.map(date => (
                    <SelectItem key={date} value={date}>
                      {new Date(date).toLocaleDateString()}
                    </SelectItem>
                  ))}
                </SelectContent>
              </Select>
            </div>
            <div>
              <label className="block text-sm font-medium mb-1">
                Stock Price
              </label>
              <div className="text-lg font-semibold">
                ${underlyingPrice.toFixed(2)}
              </div>
            </div>
            <div>
              <label className="block text-sm font-medium mb-1">
                Account Balance
              </label>
              <div className="text-lg font-semibold">
                ${balance.toLocaleString()}
              </div>
            </div>
          </div>

          {loading ? (
            <div className="text-center py-8">Loading...</div>
          ) : (
            <>
              <Table>
                <TableHeader>
                  <TableRow>
                    <TableHead>Calls</TableHead>
                    <TableHead>Bid</TableHead>
                    <TableHead>Ask</TableHead>
                    <TableHead>Volume</TableHead>
                    <TableHead>Strike</TableHead>
                    <TableHead>Bid</TableHead>
                    <TableHead>Ask</TableHead>
                    <TableHead>Volume</TableHead>
                    <TableHead>Puts</TableHead>
                  </TableRow>
                </TableHeader>
                <TableBody>
                  {optionsChain.map((option) => (
                    <TableRow key={option.strike}>
                      <TableCell>
                        <div className="space-x-2">
                          <Button 
                            size="sm"
                            variant="outline"
                            onClick={() => executeOrder(option.strike, 'call', true)}
                            disabled={!option.call.ask}
                          >
                            Buy
                          </Button>
                          <Button 
                            size="sm"
                            variant="outline"
                            onClick={() => executeOrder(option.strike, 'call', false)}
                            disabled={!option.call.bid}
                          >
                            Sell
                          </Button>
                        </div>
                      </TableCell>
                      <TableCell>{option.call.bid?.toFixed(2) || '-'}</TableCell>
                      <TableCell>{option.call.ask?.toFixed(2) || '-'}</TableCell>
                      <TableCell>{option.call.volume || '-'}</TableCell>
                      <TableCell className="font-semibold">{option.strike}</TableCell>
                      <TableCell>{option.put.bid?.toFixed(2) || '-'}</TableCell>
                      <TableCell>{option.put.ask?.toFixed(2) || '-'}</TableCell>
                      <TableCell>{option.put.volume || '-'}</TableCell>
                      <TableCell>
                        <div className="space-x-2">
                          <Button 
                            size="sm"
                            variant="outline"
                            onClick={() => executeOrder(option.strike, 'put', true)}
                            disabled={!option.put.ask}
                          >
                            Buy
                          </Button>
                          <Button 
                            size="sm"
                            variant="outline"
                            onClick={() => executeOrder(option.strike, 'put', false)}
                            disabled={!option.put.bid}
                          >
                            Sell
                          </Button>
                        </div>
                      </TableCell>
                    </TableRow>
                  ))}
                </TableBody>
              </Table>

              <div className="mt-6">
                <h3 className="text-lg font-semibold mb-2">Positions</h3>
                <Table>
                  <TableHeader>
                    <TableRow>
                      <TableHead>Type</TableHead>
                      <TableHead>Strike</TableHead>
                      <TableHead>Price</TableHead>
                      <TableHead>Side</TableHead>
                      <TableHead>Timestamp</TableHead>
                    </TableRow>
                  </TableHeader>
                  <TableBody>
                    {positions.map((position, index) => (
                      <TableRow key={index}>
                        <TableCell className="capitalize">{position.type}</TableCell>
                        <TableCell>{position.strike}</TableCell>
                        <TableCell>{position.price}</TableCell>
                        <TableCell>{position.isBuy ? 'Buy' : 'Sell'}</TableCell>
                        <TableCell>{position.timestamp}</TableCell>
                      </TableRow>
                    ))}
                  </TableBody>
                </Table>
              </div>
            </>
          )}
        </CardContent>
      </Card>
    </div>
  );
};

export default OptionsChainSimulator;
